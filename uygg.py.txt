SCALE = 100
t = {i: int(round(t_raw[i] * SCALE)) for i in t_raw}
P = [(i, i + 1) for i in range(1, 63)]
d = {j: {k: 2 * abs(j - k) for k in J} for j in J}
BIG_M = sum(t.values())

with st.sidebar:
    st.header("⚙️ Parametreler")
    L_val = st.number_input("Maks. Yürüme Mesafesi (L)", value=4)
    D_target = st.number_input("Hedef Üretim (D)", value=32)
    T_shift = st.number_input("Vardiya Süresi (dk)", value=510)
    U_max = st.slider("Maks. Doluluk (U_MAX)", 0.1, 1.0, 1.0)
    time_per_run = st.slider("Senaryo Başına Süre (sn)", 5, 60, 20)

# =========================================================
# MODEL ÇÖZÜCÜ
# =========================================================
def solve_single_scenario(worker_count):
    model = cp_model.CpModel()
    
    # Değişkenler
    x = {(i, j): model.NewBoolVar(f'x_{i}_{j}') for i in I for j in J}
    y = {(w, j): model.NewBoolVar(f'y_{w}_{j}') for w in W for j in J}
    z = {w: model.NewBoolVar(f'z_{w}') for w in W}
    s = {j: model.NewBoolVar(f's_{j}') for j in J}
    l = {j: model.NewIntVar(0, BIG_M, f'l_{j}') for j in J}
    q = {(w, j): model.NewIntVar(0, BIG_M, f'q_{w}_{j}') for w in W for j in J}
    C = model.NewIntVar(0, BIG_M, 'C')

    # Kısıtlar
    for i in I: model.Add(sum(x[i, j] for j in J) == 1)
    for i, h in P: model.Add(sum(j * x[i, j] for j in J) <= sum(j * x[h, j] for j in J))
    for j in J: model.Add(l[j] == sum(t[i] * x[i, j] for i in I))
    for j in J: model.Add(sum(y[w, j] for w in W) == s[j])
    for i in I:
        for j in J: model.Add(x[i, j] <= s[j])
    for j in J: model.Add(sum(x[i, j] for i in I) >= s[j])
    for j in range(1, len(J)): model.Add(s[j] >= s[j + 1])
    for w in W:
        for j in J: model.Add(y[w, j] <= z[w])
    for w in range(1, len(W)): model.Add(z[w] >= z[w + 1])
    
    for w in W:
        for j in J:
            model.Add(q[w, j] <= l[j])
            model.Add(q[w, j] <= BIG_M * y[w, j])
            model.Add(q[w, j] >= l[j] - BIG_M * (1 - y[w, j]))
            
    for w in W: model.Add(sum(q[w, j] for j in J) <= C)
    for j in J: model.Add(l[j] <= C)
    
    for w in W:
        model.Add(D_target * sum(q[w, j] for j in J) <= int(round(U_max * T_shift * SCALE)))
    
    for w in W:
        for j in J:
            for k in J:
                if j < k and d[j][k] > L_val: model.Add(y[w, j] + y[w, k] <= 1)
    
    # Epsilon Kısıtı: Tam olarak bu kadar işçi kullan
    model.Add(sum(z[w] for w in W) == worker_count)

    model.Minimize(C)
    
    solver = cp_model.CpSolver()
    solver.parameters.max_time_in_seconds = float(time_per_run)
    status = solver.Solve(model)

    if status in [cp_model.OPTIMAL, cp_model.FEASIBLE]:
        c_val = solver.Value(C) / SCALE
        return {
            "C": c_val,
            "output": T_shift / c_val,
            "success": "Evet" if (T_shift / c_val) >= D_target - 0.01 else "Hayır"
        }
    return None

if st.button("🚀 1-36 Tüm Operatör Senaryolarını Hesapla"):

            st.write(f"**Hedef Üretim ({D_target} adet) Sağlanıyor mu?:** {sample['Hedef?']}")
